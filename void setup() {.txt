void setup() {
  Serial.begin(9600);
  pinMode(13, OUTPUT);  // Onboard LED
  digitalWrite(13, LOW); // Start OFF
}

void loop() {
  if (Serial.available() > 0) {
    char data = Serial.read();
    Serial.print("Received: ");
    Serial.println(data);

    if (data == '1') {
      digitalWrite(13, HIGH);
    } else if (data == '0') {
      digitalWrite(13, LOW);
    }
  }
}
